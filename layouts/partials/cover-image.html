{{- /* Render a post cover image. Input dict keys: Src - cover URL under static/ (PNG covers also get AVIF/WebP sources) Title - title attribute Alt - alt text Eager - when true, render an eager, high priority image for the LCP candidate instead of a lazysizes placeholder */ -}} {{- $src := .Src -}} {{- $alt := .Alt | default .Title | default $src -}} {{- $extension := path.Ext $src -}} {{- $width := .Width -}} {{- $height := .Height -}} {{- $staticPath := printf "static%s" $src -}} {{- if and (not $width) (fileExists $staticPath) -}} {{- with imageConfig $staticPath -}} {{- $width = .Width -}} {{- $height = .Height -}} {{- end -}} {{- end -}} {{- $imgArgs := dict "Src" $src "Title" .Title "Alt" $alt "Width" $width "Height" $height -}} {{- if eq $extension ".png" -}} {{- $base := strings.TrimSuffix $extension $src -}} {{- $avif := printf "%s.avif" $base -}} {{- $webp := printf "%s.webp" $base -}} {{- if .Eager -}} {{- else -}} {{- partial "plugin/img.html" $imgArgs -}} {{- end -}} {{- else -}} {{- partial "plugin/img.html" $imgArgs -}} {{- end -}}
